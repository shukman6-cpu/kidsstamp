<script>
  // ==========================================
  // 1. 請在此處填入你向 Firebase 申請的配置資訊
  // ==========================================
  const firebaseConfig = {
    apiKey: "AIzaSyDmeRItXsjAf3PFXdsSmXa20TgYYefOIRk",
  authDomain: "kids-stamp-c378b.firebaseapp.com",
  databaseURL: "https://kids-stamp-c378b-default-rtdb.firebaseio.com",
  projectId: "kids-stamp-c378b",
  storageBucket: "kids-stamp-c378b.firebasestorage.app",
  messagingSenderId: "772769266191",
  appId: "1:772769266191:web:0f0707b8d7ba22959e417a",
  measurementId: "G-9LL6B07G6P"
};

  // 初始化 Firebase（若已配置則開啟雲端實時同步）
  let dbRef = null;
  let isFirebaseActive = false;

  if (firebaseConfig.apiKey !== "YOUR_API_KEY") {
    firebase.initializeApp(firebaseConfig);
    dbRef = firebase.database().ref('stamp_card_data');
    isFirebaseActive = true;
  }

  const defaultData = {
    elder: { name: "詩詩", wish: "去海洋公園玩一天", target: 30, avatar: "", stampImg: "", stamps: [], archives: [] },
    younger: { name: "韻韻", wish: "買一盒全新彩色筆", target: 30, avatar: "", stampImg: "", stamps: [], archives: [] }
  };

  let currentChild = 'elder';
  let appData = JSON.parse(localStorage.getItem('kids_stamp_v9')) || defaultData;

  // 儲存資料（同時同步至本地與 Firebase 雲端）
  function saveData() {
    localStorage.setItem('kids_stamp_v9', JSON.stringify(appData));
    if (isFirebaseActive) {
      dbRef.set(appData);
    }
  }

  // 監聽 Firebase 雲端資料變更（當爸爸或你在另一部手機修改時，會自動更新畫面）
  if (isFirebaseActive) {
    dbRef.on('value', (snapshot) => {
      const data = snapshot.val();
      if (data) {
        appData = data;
        localStorage.setItem('kids_stamp_v9', JSON.stringify(appData));
        document.getElementById('syncStatus').innerText = '🟢 雲端同步中（資料已是最新）';
        render();
      }
    });
  } else {
    document.getElementById('syncStatus').innerText = '⚠️ 目前使用單機模式（尚未設定 Firebase）';
  }

  function compressImage(file, callback) {
    const reader = new FileReader();
    reader.onload = function(e) {
      const img = new Image();
      img.onload = function() {
        const canvas = document.createElement('canvas');
        let width = img.width;
        let height = img.height;
        const max = 200; // 壓縮頭像/印章圖案至 200px 提升同步速度
        if (width > height && width > max) { height *= max / width; width = max; }
        else if (height > max) { width *= max / height; height = max; }
        canvas.width = width; canvas.height = height;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0, width, height);
        callback(canvas.toDataURL('image/jpeg', 0.8));
      };
      img.src = e.target.result;
    };
    reader.readAsDataURL(file);
  }

  function handleFileUpload(event, type) {
    const file = event.target.files[0];
    if (!file) return;

    compressImage(file, function(base64Img) {
      if (type === 'avatar') {
        appData[currentChild].avatar = base64Img;
      } else if (type === 'stamp') {
        appData[currentChild].stampImg = base64Img;
      }
      saveData();
      render();
    });
  }

  function switchChild(childKey) {
    currentChild = childKey;
    document.querySelectorAll('.tab-btn').forEach((btn, idx) => {
      btn.classList.toggle('active', (idx === 0 && childKey === 'elder') || (idx === 1 && childKey === 'younger'));
    });
    render();
  }

  function renderAvatar(elementId, imgData, defaultEmoji) {
    const el = document.getElementById(elementId);
    if (imgData) {
      el.innerHTML = `<img src="${imgData}" style="width:100%;height:100%;border-radius:50%;object-fit:cover;">`;
    } else {
      el.innerText = defaultEmoji;
    }
  }

  function render() {
    const data = appData[currentChild];
    
    renderAvatar('avatarElder', appData.elder.avatar, '👧');
    renderAvatar('avatarYounger', appData.younger.avatar, '👧');

    document.getElementById('wishDisplay').innerText = `願望：${data.wish}`;
    const stampsList = data.stamps || [];
    const count = stampsList.length;
    const percent = Math.min((count / data.target) * 100, 100);
    document.getElementById('progressFill').style.width = `${percent}%`;
    document.getElementById('progressText').innerText = `${count} / ${data.target} 印章`;

    const grid = document.getElementById('stampGrid');
    grid.innerHTML = '';

    for (let i = 0; i < data.target; i++) {
      const box = document.createElement('div');
      box.className = 'stamp-box';
      
      if (i < count) {
        box.classList.add('stamped');
        const stampDate = stampsList[i].date.split('/')[1] + '/' + stampsList[i].date.split('/')[2];
        if (data.stampImg) {
          box.innerHTML = `<img src="${data.stampImg}" alt="Good job"><span class="date">${stampDate}</span>`;
        } else {
          box.innerHTML = `<span style="font-size:20px;">💮</span><span class="date">${stampDate}</span>`;
        }
        box.onclick = () => onStampedClick(i);

      } else {
        if (i === count) {
          box.classList.add('next-stamp');
        }
        box.innerText = i + 1;
        box.onclick = () => onBlankClick(i);
      }

      grid.appendChild(box);
    }

    const historyList = document.getElementById('historyList');
    historyList.innerHTML = stampsList.length === 0 
      ? '<div style="color:#AAA; text-align:center; font-size:12px; padding:10px;">未有紀錄，點擊上方格子蓋第一個章！</div>'
      : [...stampsList].reverse().map(item => `<div class="history-item"><span>${item.reason}</span><span class="history-date">${item.date}</span></div>`).join('');

    const archiveList = document.getElementById('archiveList');
    const archives = data.archives || [];
    if (archives.length === 0) {
      archiveList.innerHTML = '<div style="color:#AAA; text-align:center; font-size:12px; padding:10px;">尚未有已歸檔的願望</div>';
    } else {
      archiveList.innerHTML = '';
      [...archives].reverse().forEach((arc, revIdx) => {
        const realIdx = archives.length - 1 - revIdx;
        const targetText = arc.targetStamps ? ` (${arc.totalStamps}/${arc.targetStamps}章)` : ` (${arc.totalStamps}個章)`;
        const card = document.createElement('div');
        card.className = 'archive-card';
        card.innerHTML = `
          <div>
            <div class="archive-title">🎉 ${arc.wish}${targetText}</div>
            <div style="color:#666; font-size:11px;">完成日期：${arc.completedDate}</div>
          </div>
          <div style="color:#B5A0A5; font-size:12px;">✏️</div>
        `;
        card.onclick = () => onArchiveClick(realIdx);
        archiveList.appendChild(card);
      });
    }
  }

  function onBlankClick(index) {
    const data = appData[currentChild];
    if (!data.stamps) data.stamps = [];
    if (index > data.stamps.length) {
      alert(`請先點擊第 ${data.stamps.length + 1} 個格子按順序蓋章喔！`);
      return;
    }
    
    const reason = prompt(`今天【${data.name}】做了什麼好事？`, '幫忙收拾玩具/準時完成功課');
    if (reason && reason.trim()) {
      const now = new Date();
      data.stamps.push({ date: `${now.getFullYear()}/${now.getMonth() + 1}/${now.getDate()}`, reason: reason.trim() });
      saveData();
      render();
    }
  }

  function onStampedClick(index) {
    const data = appData[currentChild];
    const item = data.stamps[index];
    
    const choice = prompt(`【第 ${index + 1} 個印章】\n當前好事：${item.reason}\n日期：${item.date}\n\n請選擇操作：\n輸入 1：修改好事內容\n輸入 2：剔走（刪除）這個印章`, '1');
    
    if (choice === '1') {
      const newReason = prompt('請修改好事內容：', item.reason);
      if (newReason && newReason.trim()) {
        item.reason = newReason.trim();
        saveData();
        render();
      }
    } else if (choice === '2') {
      if (confirm(`確定要剔走第 ${index + 1} 個印章嗎？`)) {
        data.stamps.splice(index, 1);
        saveData();
        render();
      }
    }
  }

  function onArchiveClick(index) {
    const data = appData[currentChild];
    const arc = data.archives[index];
    const targetVal = arc.targetStamps || arc.totalStamps;
    
    const choice = prompt(`【過往歷程管理】\n當前願望：${arc.wish}\n實收印章：${arc.totalStamps} 個\n目標印章：${targetVal} 個\n完成日期：${arc.completedDate}\n\n請選擇操作：\n輸入 1：修改此筆歷程資料\n輸入 2：刪除此筆歷程`, '1');
    
    if (choice === '1') {
      const newWish = prompt('修改願望名稱：', arc.wish);
      if (!newWish || !newWish.trim()) return;
      
      const newTarget = prompt('修改該次目標印章數：', targetVal);
      const newStamps = prompt('修改實際蓋印數量：', arc.totalStamps);
      const newDate = prompt('修改完成日期 (例如 2026/3/1)：', arc.completedDate);
      
      arc.wish = newWish.trim();
      if (newTarget && !isNaN(newTarget)) arc.targetStamps = parseInt(newTarget);
      if (newStamps && !isNaN(newStamps)) arc.totalStamps = parseInt(newStamps);
      if (newDate && newDate.trim()) arc.completedDate = newDate.trim();
      
      saveData();
      render();
    } else if (choice === '2') {
      if (confirm(`確定要刪除「${arc.wish}」這筆過往歷程嗎？刪除後無法恢復喔！`)) {
        data.archives.splice(index, 1);
        saveData();
        render();
      }
    }
  }

  function archiveAndNew() {
    const data = appData[currentChild];
    if (!data.stamps || data.stamps.length === 0) return alert('當前頁面還沒有任何印章！');
    if (confirm(`確定要將當前願望「${data.wish}」歸檔嗎？`)) {
      const now = new Date();
      if (!data.archives) data.archives = [];
      
      data.archives.push({ 
        wish: data.wish, 
        totalStamps: data.stamps.length, 
        targetStamps: data.target,
        completedDate: `${now.getFullYear()}/${now.getMonth() + 1}/${now.getDate()}` 
      });
      
      const newWish = prompt(`請輸入【${data.name}】的新願望：`, '新願望');
      if (newWish) {
        const newTarget = prompt(`請設定「${newWish}」的目標印章數量：`, 30);
        data.wish = newWish;
        if (newTarget && !isNaN(newTarget)) data.target = parseInt(newTarget);
      }
      
      data.stamps = [];
      saveData();
      render();
    }
  }

  function editWish() {
    const data = appData[currentChild];
    const newWish = prompt(`請輸入【${data.name}】的新願望：`, data.wish);
    if (newWish) {
      const newTarget = prompt('請設定本次願望的目標印章數量：', data.target);
      data.wish = newWish;
      if (newTarget && !isNaN(newTarget)) data.target = parseInt(newTarget);
      saveData();
      render();
    }
  }

  render();
</script>
